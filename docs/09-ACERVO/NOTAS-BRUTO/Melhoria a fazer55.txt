def detectar_categorias_multiplas(pergunta, embedding):
    resultados = []

    for cat, desc in DESCRICOES_CATEGORIA.items():
        emb_cat = gerar_embedding(desc)
        score = similaridade(embedding, emb_cat)

        resultados.append((cat, score))

    # ordena
    resultados.sort(key=lambda x: x[1], reverse=True)

    # pega TOP 3 com confiança mínima
    categorias_validas = [
        (cat, score) for cat, score in resultados[:3]
        if score > 0.30
    ]

    # fallback controlado
    if not categorias_validas:
        return [("fallback_controlado", 0.0)]

    return categorias_validas

def buscar_por_categoria(categoria, embedding):
    resultados = []

    for item in MEMORIA:
        if item['categoria'] == categoria:
            score = similaridade(embedding, item['embedding'])
            if score > 0.40:
                resultados.append({
                    **item,
                    'score_base': score
                })

    return resultados

def reranquear_candidatos(pergunta, embedding, candidatos):
    resultado = []

    termos_pergunta = extrair_termos_relevantes(pergunta)

    for item in candidatos:
        score = item['score_base']

        termos_item = extrair_termos_relevantes(item['pergunta'])

        # validação por interseção de termos
        intersecao = len(set(termos_pergunta) & set(termos_item))

        if intersecao == 0:
            score -= 0.15  # penalidade forte
        else:
            score += intersecao * 0.05

        #  penalidade por conflito semântico leve
        if conflito_semantico(pergunta, item['pergunta']):
            score -= 0.10

        # 🔹 ônus por cluster (se existir)
        if 'cluster_id' in item:
            score += 0.05

        item['score_final'] = score
        resultado.append(item)

    return sorted(resultado, key=lambda x: x['score_final'], reverse=True)

def validar_resposta(item):
    if not item:
        return False

    if item['score_final'] < 0.65:
        return False

    return True

def fallback_inteligente(pergunta, candidatos):

        return "Não encontrei nenhuma informação relacionada. Pode me ensinar?"

def extrair_termos_relevantes(texto):
    stopwords = {"o", "a", "de", "da", "do", "e", "é", "um", "uma"}

    palavras = texto.lower().split()

    return [p for p in palavras if p not in stopwords and len(p) > 3]
