import datetime

# >>> CAMINHO DOS LOGS <<<
log_path = "/content/drive/MyDrive/RAVENA/logs/log_ravena.txt"

def registrar_log(mensagem):

    agora = datetime.datetime.now()

    linha = f"{agora} - {mensagem}\n"

    with open(log_path, "a") as arquivo:
        arquivo.write(linha)

    print("LOG:", mensagem)

registrar_log("Sistema RAVENA iniciado")
