def mesma_intencao(pergunta1, pergunta2):
    palavras_q1 = set(pergunta1.lower().split())
    palavras_q2 = set(pergunta2.lower().split())

    palavras_chave_intencao = {
        "o que": "definicao",
        "como": "procedimento",
        "por que": "causa",
        "qual": "identificacao",
        "diferença": "comparacao",
        "perigos": "risco"
    }

    def extrair_intencao(p):
        for chave, tipo in palavras_chave_intencao.items():
            if chave in p:
                return tipo
        return "geral"

    return extrair_intencao(pergunta1) == extrair_intencao(pergunta2)

def overlap(p1, p2):
    s1 = set(p1.lower().split())
    s2 = set(p2.lower().split())
    return len(s1 & s2) / max(len(s1), 1)

if overlap(pergunta, candidato["pergunta"]) < 0.3:
