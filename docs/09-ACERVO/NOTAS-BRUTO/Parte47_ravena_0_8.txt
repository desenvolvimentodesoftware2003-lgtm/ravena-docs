BLOCO 3 — CONFIGURAÇÕES

MEMORIA_PATH = os.path.join(BASE_PATH, "memoria", "memoria_semantica.json") THRESHOLD = 0.65
modelo = SentenceTransformer('all-MiniLM-L6-v2')
============================================
BLOCO 4 — MEMÓRIA
============================================
def carregar_memoria(): if not os.path.exists(MEMORIA_PATH): return [] with open(MEMORIA_PATH, 'r', encoding='utf-8') as f: return json.load(f)
def salvar_memoria(memoria): os.makedirs(os.path.dirname(MEMORIA_PATH), exist_ok=True) with open(MEMORIA_PATH, 'w', encoding='utf-8') as f: json.dump(memoria, f, ensure_ascii=False, indent=4)
memoria_semantica = carregar_memoria()
============================================
BLOCO 5 — FUNÇÕES AUXILIARES
============================================
def normalizar_texto(texto): return texto.lower().strip()
def gerar_embedding(texto): return modelo.encode([texto])[0]
============================================
BLOCO 6 — BUSCA SEMÂNTICA
============================================
def buscar_memoria(pergunta): if not memoria_semantica: return None
pergunta_emb = gerar_embedding(pergunta) melhores = [] for item in memoria_semantica: similaridade = cosine_similarity( [pergunta_emb], [item['embedding']] )[0][0] melhores.append((similaridade, item)) melhores.sort(reverse=True, key=lambda x: x[0]) melhor_score, melhor_item = melhores[0] print(f"DEBUG: Similaridade: {melhor_score}") if melhor_score >= THRESHOLD: return melhor_item['resposta'] return None 
============================================
BLOCO 7 — APRENDIZADO
============================================
def aprender(pergunta, resposta): pergunta_norm = normalizar_texto(pergunta) emb = gerar_embedding(pergunta_norm)
memoria_semantica.append({ "pergunta": pergunta_norm, "resposta": resposta, "embedding": emb.tolist() }) salvar_memoria(memoria_semantica) return "Aprendido com sucesso."
