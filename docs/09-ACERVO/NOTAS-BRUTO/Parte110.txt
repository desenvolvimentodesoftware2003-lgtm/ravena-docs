def score_memoria(item):
    peso = item.get("peso", 1)
    acessos = item.get("acessos", 0)

    return (peso * 0.7) + (acessos * 0.3)

def esquecer_memorias(memoria, limite=500):

    if len(memoria) <= limite:
        return memoria

    # ordena da MENOS relevante → MAIS relevante
    memoria.sort(key=lambda x: score_memoria(x))

    # remove as piores
    excesso = len(memoria) - limite
    memoria = memoria[excesso:]

    return memoria
