def buscar_memoria(pergunta):

    pergunta_limpa = limpar_texto(pergunta)

    for chave in memoria:

        chave_limpa = limpar_texto(chave)

        if pergunta_limpa == chave_limpa:
            return memoria[chave]

    return None
