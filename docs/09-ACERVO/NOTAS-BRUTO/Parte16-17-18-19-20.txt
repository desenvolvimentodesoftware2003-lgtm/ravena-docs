def ver_memoria():

    if not memoria:
        return "A memória da RAVENA está vazia."

    texto = "Memória da RAVENA:\n\n"

    for pergunta, resposta in memoria.items():
        texto += f"{pergunta} -> {resposta}\n"

    return texto

if pergunta == "/vermemoria":
    return ver_memoria()

if pergunta == "/limparmemoria":
    return limpar_memoria()
