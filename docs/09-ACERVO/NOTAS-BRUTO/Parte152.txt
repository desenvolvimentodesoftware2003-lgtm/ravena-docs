def classificar_erro(top1, contexto):

    if top1 is None:
        return "sem_candidatos"

    if top1["item"]["resposta"] is None:
        return "erro_memoria"

    if top1["score"] < 0.75:
        return "baixa_similaridade"

    if top1["item"].get("tipo") != contexto["intencao"]:
        return "intencao_errada"

    if contexto["conf_cat"] < 0.4:
        return "categoria_fraca"

    return "desconhecido"

def corrigir_erro(tipo_erro, pergunta, candidatos):

    if tipo_erro == "erro_memoria":

        return rerank_sem_intencao(candidatos)

    if tipo_erro == "baixa_similaridade":

    return fallback_resposta(pergunta)
