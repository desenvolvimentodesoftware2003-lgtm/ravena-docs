def avaliar_aprendizado(pergunta, resposta, score, acessos=0):
    
    pontos = 0

    # 1. confiança
    if score > 0.8:
        pontos += 2
    elif score > 0.6:
        pontos += 1

    # 2. tamanho da resposta
    if len(resposta.split()) > 8:
        pontos += 1

    # 3. repetição
    if acessos > 2:
        pontos += 2

    # 4. intenção (conteúdo valioso)
    pergunta_norm = pergunta.lower()
    if any(k in pergunta_norm for k in ["o que é", "como", "por que"]):
        pontos += 1

    # 5. penalidade
    if "não sei responder" in resposta.lower():
        pontos -= 2

    return pontos

DECISÃO FINAL

def deve_aprender(pergunta, resposta, score, acessos=0):
    pontos = avaliar_aprendizado(pergunta, resposta, score, acessos)

    print(f"DEBUG_APRENDIZADO: pontos = {pontos}")

    return pontos >= 3
