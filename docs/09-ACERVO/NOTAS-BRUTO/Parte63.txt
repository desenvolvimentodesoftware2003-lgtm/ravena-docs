def classificar_topico(pergunta):
    pergunta = pergunta.lower()

    if any(p in pergunta for p in ["cigarro", "saúde", "doença", "câncer"]):
        return "saude"
    
    if any(p in pergunta for p in ["brasil", "história", "descobriu"]):
        return "historia"

    return "geral"
