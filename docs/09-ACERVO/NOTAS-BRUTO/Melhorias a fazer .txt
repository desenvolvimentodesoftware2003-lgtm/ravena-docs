# Thresholds Dinâmicos por Prioridade (O peso agora ajusta o rigor, não o score)
# Categorias com peso 1.0 exigem 0.70 de score. Categorias com 0.85 aceitam 0.60.
PESOS_DORMANCIA_CATEGORIA = {
    'ciencia': 1.0, 'tecnologia': 1.0, 'espaco': 1.0, 'matematica': 1.0,
    'saude': 0.95, 'logistica': 0.95, 'politica': 0.9, 'economia': 0.9,
    'historia': 0.85, 'geografia': 0.85, 'filosofia': 0.85, 'arte': 0.8,

# BLOCO 8 — UTILITÁRIOS DE TEXTO (REFATORADO: EXTRAÇÃO DE NÚCLEO)
# ============================================================
def extrair_nucleo_semantico(texto):
    """
    Remove o 'eu lírico' e foca no que importa: Substantivos e Verbos.
    """
    doc = nlp_pt(texto.lower())
    # Filtramos apenas o que carrega significado pesado
    tokens = [t.lemma_ for t in doc if t.pos_ in ['NOUN', 'PROPN', 'VERB', 'ADJ'] and not t.is_stop]
    return " ".join(tokens) if tokens else texto

def normalizar_texto(texto):
    if not isinstance(texto, str): return ""
    texto = remover_acentos(texto.lower().strip())
    texto = re.sub(r"[^\w\s]", "", texto)
    return extrair_nucleo_semantico(texto)

print("Gerando Mapas Conceituais das Categorias...")
CATEGORY_EMBEDDINGS = {
    cat: gerar_embedding(desc) for cat, desc in DESCRICOES_CATEGORIA.items()
}
print("CATEGORY_EMBEDDINGS prontos para uso semântico.")

# ============================================================
# BLOCO 15 — CLASSIFICAÇÃO INTELIGENTE (REFATORADO)
# ============================================================
def classificar_tipo_pergunta(pergunta_bruta):
    # 1. Extraímos o núcleo para não ser enganado por "olá", "eu queria saber sobre..."
    nucleo = extrair_nucleo_semantico(pergunta_bruta)
    emb_pergunta = gerar_embedding(nucleo)
    
    melhor_cat, max_sim = "geral", -1
    
    for cat, emb_cat in CATEGORY_EMBEDDINGS.items():
        sim = cosine_similarity([emb_pergunta], [emb_cat])[0][0]
        
        # Aplicamos o peso de importância da categoria na similaridade
        sim_ajustada = sim * PESOS_DORMANCIA_CATEGORIA.get(cat, 0.7)
        
        if sim_ajustada > max_sim:
            max_sim = sim_ajustada
            melhor_cat = cat
            
    print(f"DEBUG_RAVENA: Núcleo: '{nucleo}' -> Cat: '{melhor_cat}' (Score: {max_sim:.4f})")
    return melhor_cat
